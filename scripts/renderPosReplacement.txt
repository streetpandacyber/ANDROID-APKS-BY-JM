  const renderPos = () => (
    <View style={styles.flex}>
      <View style={styles.posHeader}>
        <View style={styles.posTitleWrap}>
          <Text style={styles.eyebrow}>POINT OF SALE</Text>
          <Text style={styles.title}>Sell faster.</Text>
        </View>
        {activeShift ? (
          <View style={styles.shiftPill}>
            <View style={styles.dot} />
            <Text style={styles.shiftPillText}>{activeShift.cashierName}</Text>
          </View>
        ) : (
          <Button
            label="Start shift"
            small
            onPress={() => {
              const name = Platform.OS === "web" ? window.prompt("Cashier name") || "" : cashier;
              const record = state.cashiers.find(c => c.name.toLowerCase() === name.trim().toLowerCase());
              const enteredPin = Platform.OS === "web" ? window.prompt(record?.pin ? "Cashier PIN" : "Shift name (optional)") || "" : cashierPin;
              if (!name) return;
              if (record?.pin && enteredPin !== record.pin) return Alert.alert("PIN required", "The cashier PIN does not match this local record.");
              const shift = { id: freshId("shift"), cashierName: name.trim(), shiftName: record?.name || undefined, startedAt: new Date().toISOString() };
              setCashier(name.trim());
              setState(s => ({ ...s, shifts: [shift, ...s.shifts], activeShiftId: shift.id }));
            }}
          />
        )}
      </View>
      <View style={styles.segment}>
        {([["sell", "Sell"], ["cart", `Cart ${cart.length ? `(${cart.length})` : ""}`], ["stock", "Stock"], ["sales", "Sales"], ["reports", "Reports"]] as [PosTab, string][]).map(([key, label]) => (
          <Pressable key={key} onPress={() => setPosTab(key)} style={[styles.segmentItem, posTab === key && styles.segmentActive]}>
            <Text style={[styles.segmentText, posTab === key && styles.segmentTextActive]}>{label}</Text>
          </Pressable>
        ))}
      </View>
      {posTab === "sell" && (
        <View style={styles.flex}>
          <ScrollView contentContainerStyle={styles.content}>
            <Card style={styles.productEditor}>
              <View style={styles.rowBetween}>
                <Text style={styles.sectionTitle}>{editingProductId ? "Edit product" : "Add product"}</Text>
                {editingProductId && <Button label="Cancel" onPress={resetProductForm} secondary small />}
              </View>
              <View style={styles.row}>
                <Field value={productName} onChangeText={setProductName} placeholder="Product name" />
                <Field value={productSku} onChangeText={setProductSku} placeholder="SKU / barcode" />
              </View>
              <View style={styles.row}>
                <Field value={productCategory} onChangeText={setProductCategory} placeholder="Category" />
                <Field value={productPrice} onChangeText={setProductPrice} placeholder="Selling price" keyboardType="decimal-pad" />
              </View>
              <View style={styles.row}>
                <Field value={productCost} onChangeText={setProductCost} placeholder="Cost price" keyboardType="decimal-pad" />
                <Field value={productStock} onChangeText={setProductStock} placeholder="Starting stock" keyboardType="decimal-pad" />
              </View>
              <View style={styles.row}>
                <Field value={productTax} onChangeText={setProductTax} placeholder="Tax %" keyboardType="decimal-pad" />
                <Field value={productThreshold} onChangeText={setProductThreshold} placeholder="Low-stock alert" keyboardType="decimal-pad" />
              </View>
              <Field value={productBarcode} onChangeText={setProductBarcode} placeholder="Barcode (optional)" />
              <Field value={productNotes} onChangeText={setProductNotes} placeholder="Product notes (optional)" multiline />
              <Text style={styles.muted}>Quantity type</Text>
              <View style={styles.row}>
                {([["unit", "Unit"], ["kg", "Kilogram"], ["liter", "Liter"]] as [typeof productQuantityType, string][]).map(([key, label]) => (
                  <Pressable key={key} onPress={() => setProductQuantityType(key)} style={[styles.chip, productQuantityType === key && styles.chipActive]}>
                    <Text style={[styles.chipText, productQuantityType === key && styles.chipTextActive]}>{label}</Text>
                  </Pressable>
                ))}
              </View>
              <Button label={editingProductId ? "Save product changes" : "Add product"} onPress={saveProduct} />
            </Card>
          </ScrollView>
          <View style={styles.searchWrap}>
            <Text style={styles.searchIcon}>⌕</Text>
            <TextInput value={query} onChangeText={setQuery} placeholder="Search products, SKU, category" placeholderTextColor={C.muted} style={styles.search} />
          </View>
          <ScrollView horizontal showsHorizontalScrollIndicator={false} contentContainerStyle={styles.chips}>
            {categories.map(c => <Pressable key={c} onPress={() => setCategory(c)} style={[styles.chip, category === c && styles.chipActive]}><Text style={[styles.chipText, category === c && styles.chipTextActive]}>{c}</Text></Pressable>)}
          </ScrollView>
          <FlatList
            data={filteredProducts}
            keyExtractor={p => p.id}
            contentContainerStyle={styles.list}
            renderItem={({ item }) => {
              const balance = item.overallStock - item.soldStock;
              const low = balance <= item.lowStockThreshold;
              return (
                <Pressable onPress={() => addToCart(item)} onLongPress={() => editProduct(item)} delayLongPress={350} style={({ pressed }) => [styles.productRow, pressed && styles.pressed]}>
                  <View style={styles.thumb}><Text style={styles.thumbText}>{item.name.slice(0, 1)}</Text></View>
                  <View style={{ flex: 1 }}><Text style={styles.productName}>{item.name}</Text><Text style={styles.muted}>{item.sku} • {item.category} • {item.quantityType}</Text></View>
                  <View style={{ alignItems: "flex-end" }}><Text style={styles.price}>{money(item.price, state.settings.currency)}</Text><Text style={[styles.stockText, low && { color: C.red }]}>{low ? `LOW • ${balance} left` : `${balance} ${item.quantityType} left`}</Text></View>
                </Pressable>
              );
            }}
            ListEmptyComponent={<View style={styles.empty}><Text style={styles.emptyTitle}>No products found</Text><Text style={styles.muted}>Try another search or category. Long-press a product row to edit it.</Text></View>}
          />

                <Text style={styles.hand}>{x.quantity} {x.product.name} = {money(x.quantity * x.unitPrice, state.settings.currency)}</Text>
                <View style={styles.qty}><IconButton label="−" onPress={() => changeQty(x.product.id, -1)} /><Text style={styles.qtyText}>{x.quantity}</Text><IconButton label="＋" onPress={() => changeQty(x.product.id, 1)} /></View>
              </View>

"LOW" : "OK"}</Text><Button label="+ Restock" small secondary onPress={() => { const qty = Platform.OS === "web" ? Number(window.prompt("Quantity to restock") || 0) : 1; if (qty > 0) setState(s => ({ ...s, products: s.products.map(x => x.id === p.id ? { ...x, overallStock: x.overallStock + qty } : x), stockAdjustments: [{ id: freshId("adjust"), productId: p.id, quantity: qty, createdAt: new Date().toISOString(), createdBy: activeShift?.cashierName || "Owner" }, ...s.stockAdjustments] })); }} /> </Card>;

C.red : item.status === "refunded" ? C.amber : C.green }}>{(item.status || "completed").toUpperCase()}</Text></Text>
